-- Number Parsers
-- Game suffix list (case matters: "qg" = e+123 but "Qg" = e+153). Index i = 10^(3*i).
local NUMBER_SUFFIXES = {
	"K", "M", "B", "T", "QA", "QI", "SX", "SP", "OC", "NO", "DC",
	"UD", "DD", "TD", "QAD", "QID", "SXD", "SPD", "OCD", "NOD", "VG",
	"UVG", "DVG", "TVG", "QAVG", "QIVG", "SXVG", "SPVG", "OCVG", "NOVG", "TG",
	"UTG", "DTG", "TTG", "QATG", "QITG", "SXTG", "SPTG", "OCTG", "NOTG", "qg",
	"Uqg", "Dqg", "Tqg", "Qaqg", "Qiqg", "Sxqg", "Spqg", "Ocqg", "Noqg", "Qg",
	"UQg", "DQg", "TQg", "QaQg", "QiQg", "SxQg", "SpQg", "OcQg", "NoQg", "Qq"
}
local SUFFIX_EXPONENT = {}
for i, name in ipairs(NUMBER_SUFFIXES) do
	SUFFIX_EXPONENT[name] = i * 3
end

local function parseFormattedNumber(str)
	if not str or str == "" then return nil end
	if type(str) == "number" then return str end

	local cleanedStr = tostring(str):gsub("<[^>]-javascript:>", ""):gsub("<[^>]->", "")
	if cleanedStr:upper():find("MAX") then
		return nil
	end

	local clean = cleanedStr:gsub("[%$%,%s]", "")

	local eNum = clean:match("([%d%.]+e%+?%d+)") or clean:match("([%d%.]+E%+?%d+)")
	if eNum then
		return tonumber(eNum)
	end

	local numStr, suffix = clean:match("^([%d%.]+)(%a*)$")
	if not numStr then return tonumber(clean) end

	local num = tonumber(numStr)
	if not num then return nil end

	-- Exact-case match first: the game's "Qg" (e+153) and "qg" (e+123) only differ by case
	if suffix and SUFFIX_EXPONENT[suffix] then
		return num * (10 ^ SUFFIX_EXPONENT[suffix])
	end

	suffix = suffix and suffix:upper() or ""
	if suffix == "" then return num end

	local multipliers = {
		K = 1e3,
		M = 1e6,
		B = 1e9,
		T = 1e12,
		Q = 1e15, QA = 1e15, QUAD = 1e15,
		QI = 1e18, QUIN = 1e18,
		SX = 1e21, SEX = 1e21, SEXT = 1e21,
		SP = 1e24, SEPT = 1e24,
		OC = 1e27, O = 1e27, OCT = 1e27,
		N = 1e30, NO = 1e30, NON = 1e30,
		D = 1e33, DC = 1e33, DEC = 1e33, DE = 1e33,
		UD = 1e36, UDE = 1e36, UDC = 1e36,
		DD = 1e39, DDE = 1e39, DDC = 1e39,
		TD = 1e42, TDE = 1e42, TDC = 1e42, TRD = 1e42,
		QAD = 1e45, QADE = 1e45,
		QID = 1e48, QIDE = 1e48,
		SXD = 1e51, SXDE = 1e51,
		SPD = 1e54, SPDE = 1e54,
		OCD = 1e57, OCDE = 1e57,
		NOD = 1e60, NODE = 1e60,
		VG = 1e63, VGE = 1e63,
		UVG = 1e66,
		DVG = 1e69,
		TVG = 1e72,
		QAVG = 1e75, QIVG = 1e78, SXVG = 1e81, SPVG = 1e84, OCVG = 1e87, NOVG = 1e90,
		TG = 1e93, UTG = 1e96, DTG = 1e99, TTG = 1e102, QATG = 1e105, QITG = 1e108,
		SXTG = 1e111, SPTG = 1e114, OCTG = 1e117, NOTG = 1e120,
		QG = 1e123, UQG = 1e126, DQG = 1e129, TQG = 1e132, QAQG = 1e135, QIQG = 1e138,
		SXQG = 1e141, SPQG = 1e144, OCQG = 1e147, NOQG = 1e150,
		QQ = 1e153, UQQ = 1e156, DQQ = 1e159, TQQ = 1e162, QAQQ = 1e165, QIQQ = 1e168,
		SXQQ = 1e171, SPQQ = 1e174, OCQQ = 1e177, NOQQ = 1e180, QQQ = 1e183
	}

	if not multipliers[suffix] then
		return nil
	end
	return num * multipliers[suffix]
end

-- How many decimals the Stats panel shows (2 -> 536.31SP). Change this to 3 for 536.310SP.
local STAT_DECIMALS = 2

-- Stats use a mantissa + exponent pair so they can go far past Lua's 1e308 limit
-- (tested range: up to e+5000). Returns m (1 <= m < 10) and e (integer), or nil.
local function toMantissaExp(raw)
	if raw == nil then return nil end

	local m, e
	if type(raw) == "number" then
		if raw ~= raw or raw == math.huge or raw == -math.huge then return nil end
		if raw == 0 then return 0, 0 end
		e = math.floor(math.log10(math.abs(raw)))
		m = raw / (10 ^ e)
	else
		local clean = tostring(raw):gsub("<[^>]->", ""):gsub("[%$%,%s]", "")

		-- Scientific text such as 9.776332776159724e+153 or 1.5E+4800 (kept as text, never a double)
		local ms, es = clean:match("^(-?[%d%.]+)[eE]([%+%-]?%d+)$")
		if ms and tonumber(ms) and tonumber(es) then
			m = tonumber(ms)
			e = math.floor(tonumber(es))
			if m == 0 then return 0, 0 end
		else
			-- Suffix text such as 536.31SP / 3.44Qg -> reuse the existing parser
			local n = parseFormattedNumber(clean)
			if not n then return nil end
			return toMantissaExp(n)
		end
	end

	-- Normalize the mantissa into [1, 10)
	local sign = m < 0 and -1 or 1
	m = math.abs(m)
	while m >= 10 do m = m / 10; e = e + 1 end
	while m < 1 do m = m * 10; e = e - 1 end
	return m * sign, e
end

local STAT_SUFFIXES = NUMBER_SUFFIXES

-- Accepts a number OR text (e.g. "9.776332776159724e+153"). Uses the letter suffixes up to
-- 1e185 (Qq); anything bigger is shown as scientific notation, e.g. 9.78e+4200.
local function formatShortNumber(val)
	local m, e = toMantissaExp(val)
	if not m then return "0" end
	if m == 0 then return "0" end

	local fmt = "%." .. STAT_DECIMALS .. "f"

	if e < 3 then
		local plain = m * (10 ^ e)
		if e >= 0 and plain == math.floor(plain) then
			return tostring(math.floor(plain))
		end
		return string.format(fmt, plain)
	end

	local groups = math.floor(e / 3)
	if groups <= #STAT_SUFFIXES then
		local scaled = m * (10 ^ (e - groups * 3))

		-- Rounding can push 999.999K up to "1000.00K"; roll that over to the next suffix
		if tonumber(string.format(fmt, scaled)) >= 1000 then
			groups = groups + 1
			if groups <= #STAT_SUFFIXES then
				scaled = m * (10 ^ (e - groups * 3))
			end
		end

		if groups <= #STAT_SUFFIXES then
			return string.format(fmt, scaled) .. STAT_SUFFIXES[groups]
		end
		-- rolled past the last suffix -> fall through to scientific notation
	end

	local mm, ee = m, e
	if tonumber(string.format(fmt, mm)) >= 10 then
		mm = 1
		ee = ee + 1
	end
	return string.format(fmt, mm) .. "e+" .. tostring(ee)
end

-- Tab System
local Tabs = {}

local function setActiveTab(tabBtn)
	for _, t in ipairs(Tabs) do
		local active = t.Button == tabBtn
		t.Section.Visible = active
		tween(t.Button, TWEEN_FAST, {
			BackgroundColor3 = active and Theme.Accent or Theme.Element,
			TextColor3 = active and Color3.fromRGB(255, 255, 255) or Theme.TextDim,
		})
		t.Stroke.Color = active and Theme.Accent or Theme.Stroke
		if active then
			updateCanvasSize(t.Section)
		end
	end
end

	local TabCorner = Instance.new("UICorner")

	TabCorner.Parent = Button

	local TabStroke = Instance.new("UIStroke")
	TabStroke.Color = Theme.Stroke

	TabStroke.Transparency = 0.35
	TabStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	TabStroke.Parent = Button

		if section.Visible then return end
		tween(Button, TWEEN_FAST, {BackgroundColor3 = Theme.ElementHover})
	end)
	Button.MouseLeave:Connect(function()
		if section.Visible then return end
		tween(Button, TWEEN_FAST, {BackgroundColor3 = Theme.Element})
	end)
	Button.MouseButton1Click:Connect(function()
		setActiveTab(Button)
	end)

	local tabData = {Button = Button, Section = section, Stroke = TabStroke}
	table.insert(Tabs, tabData)
	return tabData
end

local searchBusy = false

local function hideDropdown()
	Dropdown.Visible = false
	Dropdown.Size = UDim2.new(1, -16, 0, 0)
end

local function refreshSearch()
	if searchBusy then return end

	for _, child in ipairs(Dropdown:GetChildren()) do
		if child:IsA("TextButton") or child:IsA("TextLabel") then
			child:Destroy()
		end
	end

	local q = SearchBox.Text:lower():gsub("^%s+", ""):gsub("%s+$", "")
	if q == "" then
		hideDropdown()
		return
	end

	local matches = {}
	for _, entry in ipairs(panelIndex) do
		if entry.label:lower():find(q, 1, true) or entry.key:lower():find(q, 1, true) then
			table.insert(matches, entry)
		end
	end

	local rowH = 22
	if #matches == 0 then
		local none = Instance.new("TextLabel")
		none.Size = UDim2.new(1, 0, 0, rowH)
		none.BackgroundTransparency = 1
		none.Text = "No panels found"
		none.TextColor3 = Theme.TextDim
		none.TextSize = 11
		none.Font = Enum.Font.GothamSemibold
		none.TextXAlignment = Enum.TextXAlignment.Left
		none.LayoutOrder = 1
		none.ZIndex = 11
		none.Parent = Dropdown
		matches = {{label = nil}}
	end

	for order, entry in ipairs(matches) do
		if entry.label then
			local btn = Instance.new("TextButton")
			btn.Size = UDim2.new(1, -12, 0, rowH)
			btn.Position = UDim2.new(0, 10, 0, 0)
			btn.BackgroundColor3 = Theme.Element
			btn.BackgroundTransparency = 1
			btn.Text = entry.label

local function trackStat(statNames, displayLabel, prefix, labelColor, accentBar, hideUntilFound)
	if accentBar then
		accentBar.BackgroundColor3 = labelColor
	end
	if hideUntilFound and displayLabel.Parent then
		displayLabel.Parent.Visible = false
	end

	task.spawn(function()
		local function bindStat(obj)
			local function update()
				local rawVal
				if obj:IsA("ValueBase") then
					rawVal = obj.Value
				elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
					rawVal = obj.Text
				end

				-- Pass the raw value (number or text) so e+153...e+5000 never gets squashed into a double
				displayLabel.Text = prefix .. ": " .. formatShortNumber(rawVal)
				if hideUntilFound and displayLabel.Parent then
					displayLabel.Parent.Visible = true
				end
			end

			if obj:IsA("ValueBase") then
				obj:GetPropertyChangedSignal("Value"):Connect(update)
			elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
				obj:GetPropertyChangedSignal("Text"):Connect(update)
			end
			update()
		end

		local function findStat()
			for _, containerName in ipairs({"leaderstats", "Stats", "Values", "Data"}) do
				local container = LocalPlayer:FindFirstChild(containerName)
				if container then
					for _, name in ipairs(statNames) do
						local found = container:FindFirstChild(name)
						if found then
							bindStat(found)
							return true
						end
					end
				end
			end

			for _, name in ipairs(statNames) do
				local direct = LocalPlayer:FindFirstChild(name) or LocalPlayer:FindFirstChild(name, true)
				if direct then
					bindStat(direct)
					return true
				end
			end
			return false
		end

		while not findStat() and ScreenGui.Parent do
			task.wait(1)
		end
	end)
end

-- Cash Upgrades (Auto-detect all card IDs)
local CashUpgradesToggle = createToggle(sections.Cash, "CashUpgradesToggle", "Auto Cash Upgrades", 1)

CashUpgradesToggle.Changed = function(state)
	if state then
		task.spawn(function()
			local upgradeRemote = ReplicatedStorage:WaitForChild("Events"):WaitForChild("UpgradeTree")
			local treeCards = workspace:FindFirstChild("TreeCards") or workspace:FindFirstChild("UpgradeTreeCards") or workspace

			local function getCardStatus(id)
				local card = treeCards:FindFirstChild(tostring(id), true)
				if not card then return true, nil, false end

				local mainPart = card:FindFirstChild("MainPart")
				local surfaceGui = mainPart and mainPart:FindFirstChild("SurfaceGui")
				local hud = surfaceGui and surfaceGui:FindFirstChild("Hud")

				local maxed = false
				local price = nil
				local isMoneyRelated = false

				if hud then
					local levelLabel = hud:FindFirstChild("Level")
					if levelLabel and levelLabel:IsA("TextLabel") then
						local lvlText = levelLabel.Text:upper()
						if lvlText:find("MAX") then
							maxed = true
						else
							local current, max = lvlText:match("(%d+)%s*/%s*(%d+)")
							if current and max and tonumber(current) >= tonumber(max) then
								maxed = true
							end
						end
					end

					local priceLabel = hud:FindFirstChild("Price")
					if priceLabel and priceLabel:IsA("TextLabel") then
						price = parseFormattedNumber(priceLabel.Text)
					end

					local titleLabel = hud:FindFirstChild("Title")
					local descLabel = hud:FindFirstChild("Desc")

					local titleText = (titleLabel and titleLabel:IsA("TextLabel")) and titleLabel.Text:upper() or ""
					local descText = (descLabel and descLabel:IsA("TextLabel")) and descLabel.Text:upper() or ""

					if titleText:find("MONEY") or titleText:find("CASH") or descText:find("MONEY") or descText:find("CASH") then
						isMoneyRelated = true
					end
				end

				return maxed, price, isMoneyRelated
			end

			local function getCurrentMoney()
				local container = LocalPlayer:FindFirstChild("Stats") or LocalPlayer:FindFirstChild("leaderstats") or LocalPlayer
				local moneyObj = container:FindFirstChild("Money", true)
				if not moneyObj then return 0 end

				local val = moneyObj.Value
				if typeof(val) == "number" then
					return val
				elseif typeof(val) == "string" then
					return parseFormattedNumber(val) or 0
				end
				return 0
			end

			-- Auto-detect all card IDs from the TreeCards folder
			local function getAllCardIds()
				local ids = {}
				local cardsFolder = workspace:FindFirstChild("TreeCards") or workspace:FindFirstChild("UpgradeTreeCards")
				if cardsFolder then
					for _, child in ipairs(cardsFolder:GetChildren()) do
						local num = tonumber(child.Name)
						if num then
							table.insert(ids, num)
						end
					end
					table.sort(ids)
				end
				return ids
			end

			while CashUpgradesToggle.IsOn() and ScreenGui.Parent do
				local allMaxed = true
				local priorityList = {}
				local normalList = {}

				local allIds = getAllCardIds()
				for _, id in ipairs(allIds) do
					local maxed, price, isMoneyRelated = getCardStatus(id)
					if not maxed then
						allMaxed = false
						if isMoneyRelated then
							table.insert(priorityList, id)
						else
							table.insert(normalList, id)
						end
					end
				end

				local boughtMoneyUpgrade = false
				for _, id in ipairs(priorityList) do
					local maxed, price = getCardStatus(id)
					local currentMoney = getCurrentMoney()
					if not maxed and type(price) == "number" and price > 0 then
						if currentMoney >= price then
							upgradeRemote:FireServer(id)
							boughtMoneyUpgrade = true
							task.wait(0.15)
						end
					end
				end

				if not boughtMoneyUpgrade then
					for _, id in ipairs(normalList) do
						local maxed, price = getCardStatus(id)
						local currentMoney = getCurrentMoney()
						if not maxed and type(price) == "number" and price > 0 then
							if currentMoney >= price then
								upgradeRemote:FireServer(id)
								task.wait(0.15)
							end
						end
					end
				end

				if allMaxed then
					CashUpgradesToggle.Set(false)
					break
				end
				task.wait(0.15)
			end
		end)
	end
end

-- Machine Panel
local AutoMachineToggle = createToggle(sections.Machines, "AutoMachineToggle", "Auto Machine", 1)

AutoMachineToggle.Changed = function(state)
	if state then
		task.spawn(function()
			local machineUpgradeEvent = ReplicatedStorage:WaitForChild("Events"):WaitForChild("MachineUpgrade")
			local moreMachinesEvent = machineUpgradeEvent:WaitForChild("MoreMachines")
			local unlockMachineEvent = machineUpgradeEvent:WaitForChild("UnlockMachine")

			local listPath = workspace:WaitForChild("Boards"):WaitForChild("Machine"):WaitForChild("Info"):WaitForChild("SurfaceGui"):WaitForChild("Info"):WaitForChild("List")
			local WAIT_THRESHOLD = 30

			local rejectedMachines = {}

			local function rejectionActive()
				for _, gui in ipairs(PlayerGui:GetDescendants()) do
					if gui:IsA("TextLabel") and gui.Visible then
						if gui.Text:upper():find("NOT ENOUGH") then
							return true
						end
					end
				end
				return false
			end

			local function parseMachineNumber(text)
				local parsed = parseFormattedNumber(text)
				if type(parsed) == "number" then
					return math.floor(parsed + 0.5)
				end
				return nil
			end

			local function getPlayerMoney()
				local statsFolder = LocalPlayer:FindFirstChild("Stats")
				if statsFolder then
					local energy = statsFolder:FindFirstChild("Energy")
					if energy and energy:IsA("ValueBase") then
						if typeof(energy.Value) == "number" then
							return energy.Value
						elseif typeof(energy.Value) == "string" then
							local parsed = parseMachineNumber(energy.Value)
							if parsed then return parsed end
						end
					end
				end

				local leaderstatsFolder = LocalPlayer:FindFirstChild("leaderstats")
				if leaderstatsFolder then
					for _, child in ipairs(leaderstatsFolder:GetChildren()) do
						if child:IsA("ValueBase") then
							if typeof(child.Value) == "number" then
								return child.Value
							elseif typeof(child.Value) == "string" then
								local parsed = parseMachineNumber(child.Value)
								if parsed then return parsed end
							end
						end
					end
				end
				return 0
			end

			local function getPlayerIncome()
				local statsFolder = LocalPlayer:FindFirstChild("Stats")
				if statsFolder then
					local incomeObj = statsFolder:FindFirstChild("EnergyIncome") or statsFolder:FindFirstChild("Income") or statsFolder:FindFirstChild("Gain")
					if incomeObj and incomeObj:IsA("ValueBase") then
						if typeof(incomeObj.Value) == "number" then
							return incomeObj.Value
						elseif typeof(incomeObj.Value) == "string" then
							return parseMachineNumber(incomeObj.Value) or 0
						end
					end
				end
				return 0
			end

			local function getTextValue(parentFolder, name)
				if not parentFolder then return "N/A" end
				local obj = parentFolder:FindFirstChild(name)
				if obj then
					if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
						return obj.Text
					elseif obj:IsA("ValueBase") then
						return tostring(obj.Value)
					end
				end
				return "N/A"
			end

			while AutoMachineToggle.IsOn() and ScreenGui.Parent do
				local currentMoney = getPlayerMoney()
				local currentIncome = getPlayerIncome()

				local bestAffordableOption = nil
				local highestAffordableEfficiency = -1

				local bestWaitOption = nil
				local highestWaitEfficiency = -1

				for i = 1, 20 do
					if rejectedMachines[i] then continue end
					local machineObj = listPath:FindFirstChild("Machine" .. i)

					if machineObj then
						local infoFolder = machineObj:FindFirstChild("Info")
						local lockedFrame = infoFolder and infoFolder:FindFirstChild("Locked")
						local isLocked = lockedFrame and lockedFrame:IsA("GuiObject") and lockedFrame.Visible

						if infoFolder then
							if isLocked then
								local lockedText = getTextValue(lockedFrame, "LockedTxt")
								if lockedText == "N/A" then lockedText = getTextValue(lockedFrame, "Locked") end
								local lockedPrice = getTextValue(lockedFrame, "Price")

								local upperLocked = lockedText:upper()
								local isExpandRequirement = upperLocked:find("EXPAND") ~= nil
								local isStageRequirement = upperLocked:find("STAGE") ~= nil
								local isReachRequirement = upperLocked:find("REACH") ~= nil
								local isZoneRequirement = upperLocked:find("ZONE") ~= nil or upperLocked:find("AREA") ~= nil
								local isUnknownStatus = lockedText:find("%?%?%?") ~= nil
								local prevMachineObj = i > 1 and listPath:FindFirstChild("Machine" .. (i - 1))
								local prevInfoFolder = prevMachineObj and prevMachineObj:FindFirstChild("Info")
								local prevLockedFrame = prevInfoFolder and prevInfoFolder:FindFirstChild("Locked")
								local isPrevLocked = prevLockedFrame and prevLockedFrame:IsA("GuiObject") and prevLockedFrame.Visible

								if not isExpandRequirement and not isStageRequirement and not isReachRequirement and not isZoneRequirement and not isUnknownStatus and not isPrevLocked then
									local lockPriceVal = parseMachineNumber(lockedPrice)
									if lockPriceVal and lockPriceVal > 0 then
										local estimatedGain = (currentIncome > 0 and currentIncome or 1e6) * (i * 2)
										local eff = estimatedGain / lockPriceVal

										if currentMoney >= lockPriceVal then
											if eff > highestAffordableEfficiency then
												highestAffordableEfficiency = eff
												bestAffordableOption = {type = "unlock", index = i, price = lockPriceVal}
											end
										elseif currentIncome > 0 then
											local timeToWait = (lockPriceVal - currentMoney) / currentIncome
											if timeToWait <= WAIT_THRESHOLD and eff > highestWaitEfficiency then
												highestWaitEfficiency = eff
												bestWaitOption = {type = "unlock", index = i, price = lockPriceVal}
											end
										end
									end
								end
							else
								local price1 = getTextValue(infoFolder, "Price1")
								local price2 = getTextValue(infoFolder, "Price2")
								local gain = getTextValue(infoFolder, "Gain")

								local upgradeFrame = infoFolder:FindFirstChild("Upgrade")
								local upgradeTitle1 = getTextValue(upgradeFrame, "Title1")
								local upgradeTitle2 = getTextValue(upgradeFrame, "Title2")
								local isUpgradeMaxed = (upgradeTitle1 == "MAXED" or upgradeTitle2 == "MAXED" or price1 == "MAXED")

								local machinesFrame = infoFolder:FindFirstChild("Machines")
								local machinesTitle1 = getTextValue(machinesFrame, "Title1")
								local machinesTitle2 = getTextValue(machinesFrame, "Title2")
								local isMachinesMaxed = (machinesTitle1 == "MAXED" or machinesTitle2 == "MAXED" or price2 == "MAXED")

								local p1Val = parseMachineNumber(price1)
								local p2Val = parseMachineNumber(price2)
								local gainVal = parseMachineNumber(gain) or 1

								if not isUpgradeMaxed and p1Val and p1Val > 0 then
									local eff = gainVal / p1Val
									if currentMoney >= p1Val then
										if eff > highestAffordableEfficiency then
											highestAffordableEfficiency = eff
											bestAffordableOption = {type = "upgrade", index = i, price = p1Val}
										end
									elseif currentIncome > 0 then
										local timeToWait = (p1Val - currentMoney) / currentIncome
										if timeToWait <= WAIT_THRESHOLD and eff > highestWaitEfficiency then
											highestWaitEfficiency = eff
											bestWaitOption = {type = "upgrade", index = i, price = p1Val}
										end
									end
								end

								if not isMachinesMaxed and p2Val and p2Val > 0 then
									local eff = (gainVal / p2Val) * 1.25
									if currentMoney >= p2Val then
										if eff > highestAffordableEfficiency then
											highestAffordableEfficiency = eff
											bestAffordableOption = {type = "more", index = i, price = p2Val}
										end
									elseif currentIncome > 0 then
										local timeToWait = (p2Val - currentMoney) / currentIncome
										if timeToWait <= WAIT_THRESHOLD and eff > highestWaitEfficiency then
											highestWaitEfficiency = eff
											bestWaitOption = {type = "more", index = i, price = p2Val}
										end
									end
								end
							end
						end
					end
				end

				local shouldWait = false
				if bestWaitOption then
					if not bestAffordableOption then
						shouldWait = true
					elseif highestWaitEfficiency > (highestAffordableEfficiency * 1.3) then
						shouldWait = true
					end
				end

				if not shouldWait and bestAffordableOption then
					local chosenIndex = bestAffordableOption.index
					if not rejectedMachines[chosenIndex] then
						local clearGuard = 0
						while rejectionActive() and clearGuard < 25 do
							task.wait(0.2)
							clearGuard = clearGuard + 1
						end

						print(("[AutoMachine] Firing %s on Machine%d (price %s | money %s)"):format(
							bestAffordableOption.type,
							chosenIndex,
							tostring(bestAffordableOption.price),
							tostring(currentMoney)
						))

						if bestAffordableOption.type == "upgrade" then
							machineUpgradeEvent:FireServer(chosenIndex, true)
						elseif bestAffordableOption.type == "more" then
							moreMachinesEvent:FireServer(chosenIndex)
						elseif bestAffordableOption.type == "unlock" then
							unlockMachineEvent:FireServer(chosenIndex)
						end

						task.wait(0.4)
						if rejectionActive() then
							rejectedMachines[chosenIndex] = true
							print(("[AutoMachine] Machine%d rejected by the game -> blacklisted"):format(chosenIndex))
						end
					end
				end

				task.wait(0.3)
			end
		end)
	end
end

	if part and part:IsA("BasePart") then

	end
	return nil

local AutoUpgradeTreeToggle = createToggle(sections.Tree, "AutoUpgradeTreeToggle", "Auto Upgrade Tree", 3)

AutoUpgradeTreeToggle.Changed = function(state)
	if state then
		task.spawn(function()
			local function checkIsMaxed(progressText)
				if not progressText or progressText == "" or progressText == "N/A" then
					return false
				end

				local upperText = progressText:upper()
				if upperText:find("MAX") then
					return true
				end

				local current, maxVal = progressText:match("(%d+)%s*/%s*(%d+)")
				if current and maxVal then
					local cNum = tonumber(current)
					local mNum = tonumber(maxVal)
					if cNum and mNum and cNum >= mNum then
						return true
					end
				end

				return false
			end

			local function getCurrentLogs()
				local statsFolder = LocalPlayer:FindFirstChild("Stats")
				local logsValue = statsFolder and statsFolder:FindFirstChild("Logs")
				if logsValue then
					if typeof(logsValue.Value) == "number" then
						return logsValue.Value
					elseif typeof(logsValue.Value) == "string" then
						return parseFormattedNumber(logsValue.Value) or 0
					end
				end
				return 0
			end

			local TreeUpgradeEvent = ReplicatedStorage:FindFirstChild("Events")
				and ReplicatedStorage.Events:FindFirstChild("CutTree")
				and ReplicatedStorage.Events.CutTree:FindFirstChild("TreeUpgrade")

			while AutoUpgradeTreeToggle.IsOn() and ScreenGui.Parent do
				local currentLogs = getCurrentLogs()

				local listFolder = workspace:FindFirstChild("Boards")
					and workspace.Boards:FindFirstChild("TreeCutting")
					and workspace.Boards.TreeCutting:FindFirstChild("TreeUpgrades")
					and workspace.Boards.TreeCutting.TreeUpgrades:FindFirstChild("SurfaceGui")
					and workspace.Boards.TreeCutting.TreeUpgrades.SurfaceGui:FindFirstChild("Info")
					and workspace.Boards.TreeCutting.TreeUpgrades.SurfaceGui.Info:FindFirstChild("List")

				if listFolder and TreeUpgradeEvent then
					-- Auto-detect all Upgrade# cards instead of hardcoding 1-13
					local cards = listFolder:GetChildren()
					table.sort(cards, function(a, b)
						local na = tonumber(a.Name:match("%d+")) or math.huge
						local nb = tonumber(b.Name:match("%d+")) or math.huge
						return na < nb
					end)

					for _, upgradeItem in ipairs(cards) do
						if not AutoUpgradeTreeToggle.IsOn() then break end
						
						local i = tonumber(upgradeItem.Name:match("%d+"))
						if not i or upgradeItem.Name:sub(1, 7) ~= "Upgrade" then continue end
						
						local innerFrame = upgradeItem:FindFirstChild("Frame")

						if innerFrame then
							local lockedObj = innerFrame:FindFirstChild("Locked")
							local isLocked = lockedObj and lockedObj.Visible == true

							if not isLocked then
								local sourceProgress = innerFrame:FindFirstChild("Progress")
								local sourcePrice = innerFrame:FindFirstChild("Price")

								local progressText = sourceProgress and sourceProgress.Text or ""
								local priceText = sourcePrice and sourcePrice.Text or ""

								local isMaxed = checkIsMaxed(progressText)
								local rawPrice = parseFormattedNumber(priceText)

								if not isMaxed and rawPrice and currentLogs >= rawPrice then
									TreeUpgradeEvent:FireServer(i, true)
									task.wait(0.05)
								end
							end
						end
					end
				end
				task.wait(0.3)
			end
		end)
	end
end

local AutoMergeToggle = createToggle(sections.Grid, "AutoMergeToggle", "Auto Merge", 1)

AutoMergeToggle.Changed = function(state)
	if state then
		task.spawn(function()
			local mergeRemote = LocalPlayer:WaitForChild("GameplayEvents"):WaitForChild("Grid"):WaitForChild("MergeObject")
			local deleteRemote = LocalPlayer:WaitForChild("GameplayEvents"):WaitForChild("Grid"):WaitForChild("DeleteObject")

			while AutoMergeToggle.IsOn() and ScreenGui.Parent do
				local boards = workspace:FindFirstChild("Boards")
				local crystalGrid = boards and boards:FindFirstChild("CrystalGrid")

				if crystalGrid then
					local levelGroups = {}
					local numericLevels = {}

					for i = 1, 25 do
						local node = crystalGrid:FindFirstChild(tostring(i))
						if node then
							local gridLevel = node:FindFirstChild("GridLevel")
							local levelLabel = gridLevel and gridLevel:FindFirstChild("Level")
							if levelLabel and levelLabel:IsA("TextLabel") then
								local lvlText = levelLabel.Text
								if lvlText and lvlText ~= "" and lvlText ~= "0" and lvlText ~= "Max" then
									local numLvl = parseFormattedNumber(lvlText)
									if numLvl then
										if not levelGroups[numLvl] then
											levelGroups[numLvl] = {}
											table.insert(numericLevels, numLvl)
										end
										table.insert(levelGroups[numLvl], i)
									end
								end
							end
						end
					end

					table.sort(numericLevels)

					for _, lvlNum in ipairs(numericLevels) do
						local gridList = levelGroups[lvlNum]
						while #gridList >= 2 do
							if not AutoMergeToggle.IsOn() then break end
							local sourceGrid = table.remove(gridList)
							local targetGrid = table.remove(gridList)
							mergeRemote:FireServer(sourceGrid, targetGrid)
							task.wait(0.08)
						end
					end

					if #numericLevels > 0 then
						local lowestLvl = numericLevels[1]
						local lowestGrids = levelGroups[lowestLvl]

						local totalOccupied = 0
						for _, lvlNum in ipairs(numericLevels) do
							totalOccupied = totalOccupied + #levelGroups[lvlNum]
						end

						if #lowestGrids == 1 and totalOccupied >= 3 then
							local gridToDelete = lowestGrids[1]

							task.wait(0.1)

							local node = crystalGrid:FindFirstChild(tostring(gridToDelete))
							local gridLevel = node and node:FindFirstChild("GridLevel")
							local levelLabel = gridLevel and gridLevel:FindFirstChild("Level")
							if levelLabel and levelLabel:IsA("TextLabel") then
								local currentLvl = parseFormattedNumber(levelLabel.Text)
								if currentLvl == lowestLvl then
									deleteRemote:FireServer(gridToDelete)
								end
							end
						end
					end
				end
				task.wait(0.2)
			end
		end)
	end
end

local GridUpgradesToggle = createToggle(sections.Grid, "GridUpgradesToggle", "Grid Upgrades", 2)

GridUpgradesToggle.Changed = function(state)
	if state then
		task.spawn(function()
			local mergeUpgradesEvent = ReplicatedStorage:WaitForChild("Events"):WaitForChild("MergeUpgrades")

			local currentQuartzValue = 0
			local stats = LocalPlayer:FindFirstChild("Stats")
			if stats then
				local quartz = stats:FindFirstChild("Quartz")
				if quartz then
					if type(quartz.Value) == "number" then
						currentQuartzValue = quartz.Value
					else
						currentQuartzValue = parseFormattedNumber(tostring(quartz.Value)) or 0
					end
					quartz:GetPropertyChangedSignal("Value"):Connect(function()
						if type(quartz.Value) == "number" then
							currentQuartzValue = quartz.Value
						else
							currentQuartzValue = parseFormattedNumber(tostring(quartz.Value)) or 0
						end
					end)
				end
			end

			local trackedData = {}

			local function monitorUpgradeFrame(upgradeFrame)
				if not upgradeFrame:IsA("GuiObject") then return end
				local index = tonumber(upgradeFrame.Name:match("%d+"))
				if not index then return end

				if not trackedData[index] then
					trackedData[index] = { Price = 0, IsMaxed = false }
				end

				local function checkText(label)
					if not label or not label:IsA("TextLabel") then return end
					local rawText = label.Text
					if label.Name == "Price" then
						trackedData[index].Price = parseFormattedNumber(rawText) or 0
					elseif label.Name == "Progress" then
						local lowerText = rawText:lower()
						if lowerText:find("max") or lowerText:find("completed") then
							trackedData[index].IsMaxed = true
						else
							local current, total = rawText:match("(%d+)%s*/%s*(%d+)")
							if current and total and tonumber(current) >= tonumber(total) then
								trackedData[index].IsMaxed = true
							else
								trackedData[index].IsMaxed = false
							end
						end
					end
				end

				for _, descendant in ipairs(upgradeFrame:GetDescendants()) do
					if descendant:IsA("TextLabel") then
						checkText(descendant)
						descendant:GetPropertyChangedSignal("Text"):Connect(function()
							checkText(descendant)
						end)
					end
				end

				upgradeFrame.DescendantAdded:Connect(function(descendant)
					if descendant:IsA("TextLabel") then
						checkText(descendant)
						descendant:GetPropertyChangedSignal("Text"):Connect(function()
							checkText(descendant)
						end)
					end
				end)
			end

			local listContainer = workspace:FindFirstChild("Boards")
				and workspace.Boards:FindFirstChild("CrystalGrid")
				and workspace.Boards.CrystalGrid:FindFirstChild("GridUpgrades")
				and workspace.Boards.CrystalGrid.GridUpgrades:FindFirstChild("SurfaceGui")
				and workspace.Boards.CrystalGrid.GridUpgrades.SurfaceGui:FindFirstChild("Info")
				and workspace.Boards.CrystalGrid.GridUpgrades.SurfaceGui.Info:FindFirstChild("List")

			if listContainer then
				for _, child in ipairs(listContainer:GetChildren()) do
					monitorUpgradeFrame(child)
				end
				listContainer.ChildAdded:Connect(monitorUpgradeFrame)
			end

			while GridUpgradesToggle.IsOn() and ScreenGui.Parent do
				for index, data in pairs(trackedData) do
					if not GridUpgradesToggle.IsOn() then break end
					if not data.IsMaxed and currentQuartzValue >= data.Price and data.Price > 0 then
						mergeUpgradesEvent:FireServer(index, true)
						task.wait(0.1)
					end
				end
				task.wait(0.4)
			end

-- Jewels & Opal Panels (Shared driver - already auto-detects)
local function createBoardUpgradeToggle(parent, name, text, order, eventFolder, remoteName, boardName, upgradesName, currencyNames, singleArg)
	local toggle = createToggle(parent, name, text, order)

	toggle.Changed = function(state)
		if state then
			task.spawn(function()
				local remote = ReplicatedStorage:WaitForChild("Events"):WaitForChild(eventFolder):WaitForChild(remoteName)

				local function resolveList()
					local boards = workspace:FindFirstChild("Boards")
					local board = boards and boards:FindFirstChild(boardName)
					local upgrades = board and board:FindFirstChild(upgradesName)
					local surfaceGui = upgrades and upgrades:FindFirstChild("SurfaceGui")
					local info = surfaceGui and surfaceGui:FindFirstChild("Info")
					return info and info:FindFirstChild("List")
				end

				local listFolder = resolveList()
				while not listFolder and toggle.IsOn() and ScreenGui.Parent do
					task.wait(1)
					listFolder = resolveList()
				end
				if not listFolder then return end

				local currentCurrency = 0

				local function bindCurrency(obj)
					local function update()
						local rawVal
						if obj:IsA("ValueBase") then
							rawVal = obj.Value
						elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
							rawVal = obj.Text
						end

						if type(rawVal) == "number" then
							currentCurrency = rawVal
						else
							currentCurrency = parseFormattedNumber(tostring(rawVal)) or 0
						end
					end

					if obj:IsA("ValueBase") then
						obj:GetPropertyChangedSignal("Value"):Connect(update)
					elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
						obj:GetPropertyChangedSignal("Text"):Connect(update)
					end
					update()
				end

				local function findCurrency()
					for _, containerName in ipairs({"leaderstats", "Stats", "Values", "Data"}) do
						local container = LocalPlayer:FindFirstChild(containerName)
						if container then
							for _, currencyName in ipairs(currencyNames) do
								local found = container:FindFirstChild(currencyName)
								if found and (found:IsA("ValueBase") or found:IsA("TextLabel") or found:IsA("TextButton")) then
									bindCurrency(found)
									return true
								end
							end
						end
					end

					for _, currencyName in ipairs(currencyNames) do
						local deep = LocalPlayer:FindFirstChild(currencyName, true)
						if deep and (deep:IsA("ValueBase") or deep:IsA("TextLabel") or deep:IsA("TextButton")) then
							bindCurrency(deep)
							return true
						end
					end
					return false
				end

				if not findCurrency() then
					task.spawn(function()
						while toggle.IsOn() and not findCurrency() do
							task.wait(1)
						end
					end)
				end

				local REJECT_KEYWORDS = {"NOT ENOUGH", "CANNOT", "UNABLE", "INSUFFICIENT"}

				local function rejectionActive()
					for _, gui in ipairs(PlayerGui:GetDescendants()) do
						if gui:IsA("TextLabel") and gui.Visible then
							local t = gui.Text:upper()
							for _, kw in ipairs(REJECT_KEYWORDS) do
								if t:find(kw) then
									return true
								end
							end
						end
					end
					return false
				end

				local function isMaxedOrLocked(folder)
					local lockedObj = folder:FindFirstChild("Locked")
					if lockedObj and lockedObj:IsA("GuiObject") and lockedObj.Visible then
						return true
					end

					for _, desc in ipairs(folder:GetDescendants()) do
						if desc:IsA("TextLabel") or desc:IsA("TextButton") then
							if desc.Text:upper():find("MAX") then
								return true
							end
							local current, maxVal = desc.Text:match("(%d+)%s*/%s*(%d+)")
							if current and maxVal and tonumber(current) >= tonumber(maxVal) then
								return true
							end
						end
					end
					return false
				end

				local function getPrice(folder)
					local holders = {folder}
					local innerFrame = folder:FindFirstChild("Frame")
					if innerFrame then
						table.insert(holders, innerFrame)
					end

					for _, holder in ipairs(holders) do
						for _, priceName in ipairs({"Price", "Cost"}) do
							local priceObj = holder:FindFirstChild(priceName)
							if priceObj then
								if priceObj:IsA("TextLabel") or priceObj:IsA("TextButton") then
									local parsed = parseFormattedNumber(priceObj.Text)
									if parsed then return parsed end
								elseif priceObj:IsA("ValueBase") then
									if typeof(priceObj.Value) == "number" then
										return priceObj.Value
									end
									local parsed = parseFormattedNumber(tostring(priceObj.Value))
									if parsed then return parsed end
								end
							end
						end
					end
					return nil
				end

				local blockedUntil = {}

				while toggle.IsOn() and ScreenGui.Parent do
					local cards = listFolder:GetChildren()
					table.sort(cards, function(a, b)
						local na = tonumber(a.Name:match("%d+")) or math.huge
						local nb = tonumber(b.Name:match("%d+")) or math.huge
						return na < nb
					end)

					for _, item in ipairs(cards) do
						if not toggle.IsOn() then break end

						local idx = tonumber(item.Name:match("%d+"))
						if not idx or item.Name:sub(1, 7) ~= "Upgrade" then continue end
						if blockedUntil[idx] and os.clock() < blockedUntil[idx] then continue end
						if isMaxedOrLocked(item) then continue end

						local price = getPrice(item)
						if not price or price <= 0 or currentCurrency < price then continue end

						local priceObj = item:FindFirstChild("Price")
						local levelObj = item:FindFirstChild("Level")
						local beforePrice = (priceObj and (priceObj:IsA("TextLabel") or priceObj:IsA("TextButton"))) and priceObj.Text or nil
						local beforeLevel = (levelObj and (levelObj:IsA("TextLabel") or levelObj:IsA("TextButton"))) and levelObj.Text or nil

						local clearGuard = 0
						while rejectionActive() and clearGuard < 25 do
							task.wait(0.2)
							clearGuard = clearGuard + 1
						end

						if singleArg then
							remote:FireServer(idx)
						else
							remote:FireServer(idx, true)
						end

						task.wait(0.5)

						local changed = false
						if beforePrice and priceObj.Text ~= beforePrice then
							changed = true
						end
						if beforeLevel and levelObj.Text ~= beforeLevel then
							changed = true
						end

						if changed then
							task.wait(0.15)
						elseif rejectionActive() then
							blockedUntil[idx] = os.clock() + 90
							print(("[%s] Upgrade%d rejected by the game -> retry in 90s"):format(text, idx))
						else
							blockedUntil[idx] = os.clock() + 60
							print(("[%s] Upgrade%d no change after fire -> retry in 60s"):format(text, idx))
						end
					end

					task.wait(0.3)
				end
			end)
		end
	end

	return toggle
end

-- Board layout (from Explorer): <Board>.SurfaceGui.Info.List -> U1, U2, U3, ...
-- Reads the direct children of every "List" under the board's SurfaceGui(s),
-- preferring SurfaceGui, and returns cards sorted by their number (no max hardcoded).
local function collectGoldenCards(board)
	local lists = {}

	local function addList(gui)
		local info = gui and gui:FindFirstChild("Info")
		local list = info and info:FindFirstChild("List")
		if list then table.insert(lists, list) end
	end

	addList(board:FindFirstChild("SurfaceGui"))
	for _, child in ipairs(board:GetChildren()) do
		if child:IsA("SurfaceGui") and child.Name ~= "SurfaceGui" then
			addList(child)
		end
	end

	local byIdx = {}
	for _, list in ipairs(lists) do
		for _, obj in ipairs(list:GetChildren()) do
			local n = obj.Name:match("^U(%d+)$") or obj.Name:match("^Upgrade(%d+)$")
			n = tonumber(n)
			if n and not byIdx[n] then
				byIdx[n] = obj
			end
		end
		if next(byIdx) then break end -- use the first list that actually has cards
	end

	local result = {}
	for idx, obj in pairs(byIdx) do
		table.insert(result, {idx = idx, obj = obj})
	end
	table.sort(result, function(x, y) return x.idx < y.idx end)
	return result
end

-- Card layout: Icon, Gain, Price, PriceIcon, Level, Button, Design (+ UIScale/UIAspectRatio)
local function goldenText(card, childName)
	local o = card:FindFirstChild(childName)
	if o and (o:IsA("TextLabel") or o:IsA("TextButton")) then
		return o.Text
	end
	return nil
end

local function goldenCardMaxedOrLocked(card)
	local lockedObj = card:FindFirstChild("Locked")
	if lockedObj and lockedObj:IsA("GuiObject") and lockedObj.Visible then
		return true
	end

	for _, childName in ipairs({"Level", "Price", "Button"}) do
		local t = goldenText(card, childName)
		if t then
			local up = t:upper()
			if up:find("MAX") or up:find("LOCK") then
				return true
			end
			if childName == "Level" then
				local cur, mx = t:match("(%d+)%s*/%s*(%d+)")
				if cur and mx and tonumber(cur) >= tonumber(mx) then
					return true
				end
			end
		end
	end
	return false
end

local function goldenCardPrice(card)
	local holders = {card}
	local inner = card:FindFirstChild("Frame")
	if inner then table.insert(holders, inner) end

	for _, holder in ipairs(holders) do
		for _, priceName in ipairs({"Price", "Cost"}) do
			local p = holder:FindFirstChild(priceName)
			if p then
				if p:IsA("TextLabel") or p:IsA("TextButton") then
					return parseFormattedNumber(p.Text)
				elseif p:IsA("ValueBase") then
					if typeof(p.Value) == "number" then return p.Value end
					return parseFormattedNumber(tostring(p.Value))
				end
			end
		end
	end
	return nil
end

local function goldenCardSignature(card)
	return table.concat({
		goldenText(card, "Level") or "",
		goldenText(card, "Price") or "",
		goldenText(card, "Gain") or "",
	}, "|")
end

				local names = {}
				for _, c in ipairs(goldenBoard:GetChildren()) do table.insert(names, c.Name) end
				print(("[Golden] '%s' not found. Children of GoldenResearch: %s"):format(boardName, table.concat(names, ", ")))
				return
			end
			print(("[Golden] Using %s"):format(board:GetFullName()))

			local cooldown = {}
			local warned = false

			while toggle.IsOn() and ScreenGui.Parent do
				tryBindGoldenCurrency()

				local cards = collectGoldenCards(board)
				if #cards == 0 and not warned then
					warned = true
					print(("[Golden] No U# cards found under Boards.GoldenResearch.%s.SurfaceGui.Info.List"):format(boardName))
				end

				for _, entry in ipairs(cards) do
					if not toggle.IsOn() then break end

					local idx, card = entry.idx, entry.obj
					if cooldown[idx] and os.clock() < cooldown[idx] then continue end
					if goldenCardMaxedOrLocked(card) then continue end

					local price = goldenCardPrice(card)
					if price and price > 0 and GoldenCurrency.bound and GoldenCurrency.value < price then
						continue
					end

					local before = goldenCardSignature(card)
					remote:FireServer(makeArgs(idx))
					task.wait(0.3)

					if goldenCardSignature(card) ~= before then
						table.clear(cooldown)
					else
						cooldown[idx] = os.clock() + 5
					end
				end

				task.wait(0.2)
			end
		end)
	end

	return toggle
end

-- Perks Panel (Already auto-detects)
local PerksToggle = createToggle(sections.Perks, "PerksToggle", "Auto Perk Upgrades", 1)

PerksToggle.Changed = function(state)
	if state then
		task.spawn(function()
			local perkRemote = ReplicatedStorage:WaitForChild("Events"):WaitForChild("PerkUpgrade")

			local listFolder = workspace:WaitForChild("Boards")
				:WaitForChild("Levels")
				:WaitForChild("PerksTree")
				:WaitForChild("SurfaceGui")
				:WaitForChild("Tree")
				:WaitForChild("List")

			local currentPerkCurrency = 0

			local function bindCurrencyObject(obj)
				local function update()
					local rawVal
					if obj:IsA("ValueBase") then
						rawVal = obj.Value
					elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
						rawVal = obj.Text
					end

					if type(rawVal) == "number" then
						currentPerkCurrency = rawVal
					else
						currentPerkCurrency = parseFormattedNumber(tostring(rawVal)) or 0
					end
				end

				if obj:IsA("ValueBase") then
					obj:GetPropertyChangedSignal("Value"):Connect(update)
				elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
					obj:GetPropertyChangedSignal("Text"):Connect(update)
				end
				update()
			end

			local function findPerkCurrency()
				for _, containerName in ipairs({"leaderstats", "Stats", "Values", "Data"}) do
					local container = LocalPlayer:FindFirstChild(containerName)
					if container then
						for _, name in ipairs({"PerkPoints", "Perks", "Perk", "PP", "SkillPoints", "Points"}) do
							local found = container:FindFirstChild(name)
							if found then
								bindCurrencyObject(found)
								return true
							end
						end
					end
				end

				for _, name in ipairs({"PerkPoints", "Perks", "Perk", "PP", "SkillPoints", "Points"}) do
					local deep = LocalPlayer:FindFirstChild(name, true)
					if deep then
						bindCurrencyObject(deep)
						return true
					end
				end
				return false
			end

			if not findPerkCurrency() then
				task.spawn(function()
					while PerksToggle.IsOn() and not findPerkCurrency() do
						task.wait(1)
					end
				end)
			end

			local REJECT_KEYWORDS = {"NOT ENOUGH", "CANNOT", "UNABLE", "INSUFFICIENT"}

			local function rejectionActive()
				for _, gui in ipairs(PlayerGui:GetDescendants()) do
					if gui:IsA("TextLabel") and gui.Visible then
						local t = gui.Text:upper()
						for _, kw in ipairs(REJECT_KEYWORDS) do
							if t:find(kw) then
								return true
							end
						end
					end
				end
				return false
			end

			local function isPerkMaxed(folder)
				for _, desc in ipairs(folder:GetDescendants()) do
					if desc:IsA("TextLabel") or desc:IsA("TextButton") then
						if desc.Text:upper():find("MAX") then
							return true
						end
					end
				end
				return false
			end

			local LOCK_KEYWORDS = {"LOCKED", "LOCK", "REQUIRES", "UNLOCK AT", "UNLOCKS AT"}

			local function isPerkLocked(folder)
				-- Explicit "Locked" instance (BoolValue/Frame/etc.) some perk trees use
				local lockedFlag = folder:FindFirstChild("Locked")
				if lockedFlag then
					if lockedFlag:IsA("BoolValue") then
						if lockedFlag.Value then return true end
					elseif lockedFlag:IsA("GuiObject") then
						if lockedFlag.Visible then return true end
					end
				end

				for _, desc in ipairs(folder:GetDescendants()) do
					if desc:IsA("TextLabel") or desc:IsA("TextButton") then
						local t = desc.Text:upper()
						for _, kw in ipairs(LOCK_KEYWORDS) do
							if t:find(kw) then
								return true
							end
						end
					elseif desc:IsA("ImageLabel") or desc:IsA("ImageButton") then
						-- Common lock-icon naming convention
						if desc.Name:upper():find("LOCK") and desc.Visible then
							return true
						end
					end
				end
				return false
			end

			local function getPerkPrice(folder)
				local priceObj = folder:FindFirstChild("Price")
				if priceObj then
					if priceObj:IsA("TextLabel") or priceObj:IsA("TextButton") then
						return parseFormattedNumber(priceObj.Text)
					elseif priceObj:IsA("ValueBase") then
						if typeof(priceObj.Value) == "number" then
							return priceObj.Value
						end
						return parseFormattedNumber(tostring(priceObj.Value))
					end
				end
				return nil
			end

			local blockedUntil = {}
			local lockedUntil = {}

			while PerksToggle.IsOn() and ScreenGui.Parent do
				local cards = listFolder:GetChildren()
				table.sort(cards, function(a, b)
					local na = tonumber(a.Name:match("%d+")) or math.huge
					local nb = tonumber(b.Name:match("%d+")) or math.huge
					return na < nb
				end)

				for _, item in ipairs(cards) do
					if not PerksToggle.IsOn() then break end

					local idx = tonumber(item.Name:match("%d+"))
					if not idx or item.Name:sub(1, 7) ~= "Upgrade" then continue end
					if blockedUntil[idx] and os.clock() < blockedUntil[idx] then continue end
					if lockedUntil[idx] and os.clock() < lockedUntil[idx] then continue end

					if isPerkMaxed(item) then continue end

					-- Locked perks (waiting on a prerequisite) are never fired at the
					-- server and never get the long rejection cooldown, so they get
					-- re-checked cheaply and buy the instant they unlock instead of
					-- getting starved behind later upgrades.
					if isPerkLocked(item) then
						lockedUntil[idx] = os.clock() + 2
						continue
					end

					local price = getPerkPrice(item)
					if not price or price <= 0 or currentPerkCurrency < price then continue end

					local priceObj = item:FindFirstChild("Price")
					local gainObj = item:FindFirstChild("Gain")
					local beforePrice = (priceObj and (priceObj:IsA("TextLabel") or priceObj:IsA("TextButton"))) and priceObj.Text or nil
					local beforeGain = (gainObj and (gainObj:IsA("TextLabel") or gainObj:IsA("TextButton"))) and gainObj.Text or nil

					local clearGuard = 0
					while rejectionActive() and clearGuard < 25 do
						task.wait(0.2)
						clearGuard = clearGuard + 1
					end

					perkRemote:FireServer(idx, true)

					task.wait(0.5)

					local changed = false
					if beforePrice and priceObj.Text ~= beforePrice then
						changed = true
					end
					if beforeGain and gainObj.Text ~= beforeGain then
						changed = true
					end

					if changed then
						-- A purchase went through: game state changed, so any earlier
						-- upgrade that was previously blocked/locked deserves an
						-- immediate re-check next pass instead of sitting out a stale
						-- cooldown while currency keeps flowing to later upgrades.
						table.clear(blockedUntil)
						table.clear(lockedUntil)
						task.wait(0.15)
					elseif rejectionActive() then
						blockedUntil[idx] = os.clock() + 20
						print(("[Perks] Upgrade%d rejected by the game -> retry in 20s"):format(idx))
					else
						blockedUntil[idx] = os.clock() + 10
						print(("[Perks] Upgrade%d no change after fire -> retry in 10s"):format(idx))
					end
				end

				task.wait(0.3)
			end
		end)
	end
end

		if typeof(firetouchinterest) ~= "function" then
			warn("[Buttons] firetouchinterest is not available in this executor.")

-- Dragging
local dragging = false
local dragInput = nil
local dragStart = nil
local startPos = nil

Header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = MainFrame.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

Header.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if input == dragInput and dragging then
		local delta = input.Position - dragStart
		local newPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		MainFrame.Position = newPos
		Shadow.Position = UDim2.new(newPos.X.Scale, newPos.X.Offset + 2, newPos.Y.Scale, newPos.Y.Offset + 6)
	end
end)

-- Minimize
local minimized = false
MinimizeBtn.MouseButton1Click:Connect(function()
	minimized = not minimized
	local targetSize = minimized and UDim2.new(0, 280, 0, 32) or UDim2.new(0, 280, 0, 346)
	local shadowTarget = minimized and UDim2.new(0, 284, 0, 36) or UDim2.new(0, 284, 0, 352)

	if minimized then
		PanelContainer.Visible = false
		Content.Visible = false
		SearchRow.Visible = false
		MasterBtn.Visible = false
		hideDropdown()
		MinimizeBtn.Text = "+"
	end

	local tweenInfo = TweenInfo.new(0.28, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
	local sizeTween = TweenService:Create(MainFrame, tweenInfo, {Size = targetSize})
	TweenService:Create(Shadow, tweenInfo, {Size = shadowTarget}):Play()

	if not minimized then
		sizeTween.Completed:Connect(function()
			if not minimized then
				PanelContainer.Visible = true
				Content.Visible = true
				SearchRow.Visible = true
				MasterBtn.Visible = true
				MinimizeBtn.Text = "_"
			end
		end)
	end

	sizeTween:Play()
end)
